  /**
   *  Prints a warning and asks if to continue
   *
   * @param  text  The warning text
   * @param  br    The buffered reader from System.in
   * @return       True=continue, false=break
   */
  private static boolean warn(String text, BufferedReader br) {
    System.out.println(text + ". Do you want to continue (y/n) ?");
    String action = null;
    try {
      action = br.readLine();
    } catch (IOException ioe) {
      System.out.println("IO error trying to read your name!");
      System.exit(1);
    }
    if (action.equals("y")) {
      return (true);
    } else {
      return (false);
    }
  }


  /**
   *  Executes reading Gaussain log files
   *
   * @param  args  The command line args
   */
  private static void executeReadGaussian(String[] args) throws Exception{
	  Torque.init(args[1]+"../conf/jetspeed/WEB-INF/conf/Torque.properties");
	  TurbineConfig config = new TurbineConfig(args[1], "../conf/jetspeed/WEB-INF/conf/TurbineResources.properties");
	  config.initialize();
	  G98Shifts g98 = new G98Shifts();
	  String[] ht = {"--carbonref", "189.70", "--protonref", "32.18"};
	  File[] files = new File(args[0]).listFiles();
	  for (int i = 0; i < files.length; i++) {
		  if (files[i].getName().substring(files[i].getName().length() - 4).equals(".log")) {
			  AbInitioDataset aid = null;
			  System.out.println("Starting to read: " + files[i].getName());
			  try {
				  aid = g98.readG9x(files[i], ht);
				  doInsert(aid, args);
				  System.out.println("Succesfully written: " + files[i].getName() + ", " + i + ". file");
				  files[i].delete();
				  System.out.println("deleted");
			  } catch (Exception ex) {
				  System.out.println("Failed to write: " + files[i].getName() + ", because of " + ex.getMessage());
			  }
		  }
	  }
  }


  /**
   *  Executes reading dkfz files
   *
   * @param  args           The command line args
   * @exception  Exception  Description of Exception
   */
  private static void executeReadDkfz(String[] args) throws Exception {
	  Torque.init(args[1]+"../conf/jetspeed/WEB-INF/conf/Torque.properties");
	  TurbineConfig config = new TurbineConfig(args[1], "../conf/jetspeed/WEB-INF/conf/TurbineResources.properties");

	  String geraetepropsFile = args[0] + "/geraetemod.txt";
	  Properties geraeteprops = new Properties();
	  geraeteprops.load(new FileInputStream(geraetepropsFile));
	  NmrshiftdbUser user = null;
	  Criteria crit = new Criteria();

	  } else {

		  user = new NmrshiftdbUser();

		  user.setPasswordValue("dkfz");
		  user.setCreated(new Date());
		  user.setLastLogin(new Date(0));
		  user.setFirstName("dkfz");
		  user.setLastName("dkfz");
		  user.setTitle("dkfz");
		  user.setAddress("dkfz");
		  user.setCity("dkfz");
		  user.setState("dkfz");
		  user.setZipCode("dkfz");
		  user.setCountry("dkfz");
		  user.setWebPage("dkfz");
		  user.setAffiliation1("dkfz");
		  user.setAffiliation2("dkfz");
		  user.setEmail("dkfz");
		  user.setHallOfFame("false");
		  user.save();
	  }
	  String nmrpropsFile = args[1] + "../conf/jetspeed/WEB-INF/conf/NMRShiftDB.properties";
	  String jetspeedpropsFile = args[1] + "../conf/jetspeed/WEB-INF/conf/JetspeedResources.properties";
	  Properties nmrprops = new Properties();
	  Properties jetspeedprops = new Properties();
	  nmrprops.load(new FileInputStream(nmrpropsFile));
	  jetspeedprops.load(new FileInputStream(jetspeedpropsFile));
	  for (int i = 0; i < files.length; i++) {
		  try {
			  if (files[i].getName().substring(files[i].getName().length() - 7).equals(".SPDATA")) {
				  BufferedReader buffRead = new BufferedReader(new FileReader(files[i]));
				  String line = "";
				  TreeMap map = new TreeMap();
				  int counter = 0;
				  while ((line = buffRead.readLine()) != null) {
					  if (line.indexOf("$$$$") != -1) {
						  try {
							  processData(map, args, geraeteprops, user, nmrprops, jetspeedprops);
						  } catch (Exception ex) {
							  ex.printStackTrace();
						  }
						  map = new TreeMap();
						  counter = 0;
					  } else {
						  map.put(line.substring(0, 4).trim() + counter++, line.substring(line.length()-1).equals("<") ? line.substring(4,line.length() - 1).trim() : line.substring(4,line.length()).trim());
					  }
				  }
			  }
		  } catch (Exception ex) {
			  ex.printStackTrace();
		  }
	  }
  }

  /**
   *  Inserts a dkfz set into the db
   *
   * @param  map            The data as a hashmap
   * @param  args           The command line args
   * @param  geraeteprops   Description of Parameter
   * @param  user           Description of Parameter
   * @param  nmrprops       Description of Parameter
   * @param  jetspeedprops  Description of Parameter
   * @exception  Exception  Databse problems
   */
  private static void processData(Map map, String[] args, Properties geraeteprops, NmrshiftdbUser user, Properties nmrprops, Properties jetspeedprops) throws Exception {
	  SubmittingData subdata = new SubmittingData(args[1] + "../conf/normalizer.xml");
	  System.out.println("processing " + map.get("$ID0"));
	  //Read the molcule
	  int counter = 0;
	  boolean after = false;
	  boolean before = true;
	  IMolecule mol = new org.openscience.cdk.Molecule();
	  while (!after) {
		  String x = (String) map.get("$CT" + counter);
		  counter++;
		  if (x != null) {
			  before = false;
			  StringTokenizer st = new StringTokenizer(x);
			  boolean addAtom = true;
			  int a = 0;
			  IAtom atom = null;
			  int numberofbonds = 0;
			  while (st.hasMoreTokens()) {
				  String line = st.nextToken();
				  if (addAtom) {
					  line = st.nextToken();
					  addAtom = false;
					  IIsotope i = IsotopeFactory.getInstance(DefaultChemObjectBuilder.getInstance()).getMajorIsotope(Integer.parseInt(line));
					  atom = new org.openscience.cdk.Atom(i.getSymbol());
					  IsotopeFactory.getInstance(DefaultChemObjectBuilder.getInstance()).configure(atom, i);
					  mol.addAtom(atom);
					  a = mol.getAtomNumber(atom);
					  line = st.nextToken();
				  } else {
					  if (numberofbonds == 6) {
						  atom.getPoint2d().x=Integer.parseInt(line);
						  String line2 = st.nextToken();
						  atom.getPoint2d().y=Integer.parseInt(line2);
					  } else {
						  String line2 = st.nextToken();
						  numberofbonds++;
						  int atomnumber = Integer.parseInt(line) - 1;
						  if (atomnumber != -1) {
							  if (atomnumber < mol.getAtomCount()) {
								  int bondorder = Integer.parseInt(line2);
								  if (bondorder < 4) {
									  mol.addBond(a, atomnumber, AtomUtils.getOrderFromInt(bondorder));
								  }
								  if (bondorder == 4) {
									  IBond bond = new org.openscience.cdk.Bond(atom, mol.getAtom(atomnumber), IBond.Order.SINGLE);
									  bond.setFlag(CDKConstants.ISAROMATIC,true);
									  mol.addBond(bond);
									  atom.setFlag(CDKConstants.ISAROMATIC,true);
									  mol.getAtom(atomnumber).setFlag(CDKConstants.ISAROMATIC,true);
								  }
							  }
						  }
					  }
				  }
			  }
		  } else {
			  if (!before) {
				  after = true;
			  }
		  }
	  }
	  StringWriter sw = new StringWriter();
	  AtomUtils.addAndPlaceHydrogens(mol);
	  SaturationChecker satcheck = new SaturationChecker();
	  satcheck.saturate(mol);
	  new MDLWriter(sw).writeMolecule((IMolecule)AtomContainerManipulator.removeHydrogens(mol));
	  subdata.setMol(sw.toString(),false,true,false);
	  //read shifts
	  after = false;
	  before = true;
	  int counter2 = 0;
	  int numberOfSignals = 0;
	  while (!after) {
		  String x = (String) map.get("$LX" + counter2);
		  counter2++;
		  if (x != null) {
			  before = false;
			  StringTokenizer st = new StringTokenizer(x);
			  while (st.hasMoreTokens()) {
				  subdata.addToSignalstable(new ValueTriple(Float.parseFloat(st.nextToken())));
				  numberOfSignals++;
			  }
		  } else {
			  if (!before) {
				  after = true;
			  }
		  }
	  }
	  //read rest
	  Iterator iterator = map.keySet().iterator();
	  Vector allAssignments = new Vector();
	  String intensity = "";
	  String chemicalName = "";
	  String reference = "DKFZ \'spektren\' database: ";
	  String ex = "";
	  String kommentar = "";
	  String solvent = "";
	  String lz1="";
	  String lz2="";
	  String lz3="";
	  String lz4="";
	  while (iterator.hasNext()) {
		  String element = (String) iterator.next();
		  if (element.indexOf("$LZ1") != -1){
			  lz1+=map.get(element)+"   ";
		  }
		  if (element.indexOf("$LZ2") != -1){
			  lz2+=map.get(element)+"   ";
		  }
		  if (element.indexOf("$LZ3") != -1){
			  lz3+=map.get(element)+"   ";
		  }
		  if (element.indexOf("$LZ4") != -1){
			  lz4+=map.get(element)+"   ";
		  }
		  if (element.indexOf("$NA") != -1) {
			  chemicalName += ((String) map.get(element));
		  }
		  if (element.indexOf("$HE") != -1 || element.indexOf("$RF") != -1) {
			  reference += ((String) map.get(element)) + "/";
		  }
		  if (element.indexOf("$LY") != -1) {
			  intensity += ((String) map.get(element)) + "    ";
		  }
		  if (element.indexOf("$EX") != -1) {
			  ex += ((String) map.get(element)) + "    ";
		  }
		  if (element.indexOf("$KO") != -1) {
			  kommentar += ((String) map.get(element)) + "    ";
		  }
		  if (element.indexOf("$LM") != -1) {
			  solvent += ((String) map.get(element)).trim();
		  }
	  }
	  if(!lz1.equals("")){
		  allAssignments.add(lz1);
	  }
	  if(!lz2.equals("")){
		  allAssignments.add(lz2);
	  }
	  if(!lz3.equals("")){
		  allAssignments.add(lz3);
	  }
	  if(!lz4.equals("")){
		  allAssignments.add(lz4);
	  }
	  for(int i=0;i<allAssignments.size();i++){
		  String assignment=(String)allAssignments.get(i);
		  int counter3 = 0;
		  StringTokenizer st = new StringTokenizer(assignment);
		  while (st.hasMoreTokens()) {
			  if( subdata.getSignalstable().size()<=counter3){
				  System.out.println("File corrupt. More assignments than signals!");
				  break;
			  }
			  String x = st.nextToken();
			  if (!x.equals("0")) {
				  ((ValueTriple) subdata.getSignalstable().get(counter3)).atoms.add(subdata.getMolWithH().getAtom(Integer.parseInt(x) - 1));
			  }
			  counter3++;
		  }
	  }
	  StringTokenizer st = new StringTokenizer(intensity);
	  boolean all100 = true;
	  Vector intensities = new Vector();
	  while (st.hasMoreTokens()) {
		  String x = st.nextToken();
		  Float d = new Float(x);
		  intensities.add(d);
		  if (d.floatValue() != 100) {
			  all100 = false;
		  }
	  }
	  if (!all100) {
		  float max = 0;
		  for (int i = 0; i < intensities.size(); i++) {

		  }

			  if (mhz == null) {
				  ((DBConditionType) subdata.getConditions().get(i)).setCurrentValue(new String("Unreported"));

				  ((DBConditionType) subdata.getConditions().get(i)).setCurrentValue(new String(Integer.parseInt(mhz) / 100 + ""));

				  }
			  }
		  } catch (Exception e) {}
	  }
	  for (int i = 0; i < subdata.getConditions().size(); i++) {
		  if (((DBConditionType) subdata.getConditions().get(i)).getConditionName().equals("Temperature [K]")) {

			  } else {

		  FileInputStream fstreamout = new FileInputStream(outputfile);

			  float shift=shifts.get(l).floatValue();
			  System.err.println((value)+"  "+(shift));
			  if(((int)(value*10))==((int)(shift*10)))
				  hits[i]++;
			  sqerror+=(value-shift)*(value-shift);
		  }

		  in.close();
		  outputfile.delete();
		  modelfile.delete();
	  }
	  System.err.println("hits average :"+Statistics.average(hits));
	  System.err.println("mse "+Statistics.average(errors));
  }

  private static void executeCheckNCI(String[] args) throws Exception {

	  int ncicount=0;
	  int[] found=new int[7];
	  int[] notfound=new int[7];
	  int wishedstructures=0;

		  boolean wishit=false;

		  mol=(IMolecule)AtomContainerManipulator.removeHydrogens(mol);
	        SmilesGenerator sg = new SmilesGenerator();

		  List v=DBMoleculePeer.doSelectBySmiles(smiles, null, true);
		  if(v.size()>0){
			  DBMolecule dbmol=(DBMolecule)v.get(0);

			  IAtom a=mol.getAtom(i);
			  for(int k=1;k<7;k++){
				  if(a.getSymbol().equals("C")){
					  String hosecode=hcg.getHOSECode(mol,a,k,true);
					  StringBuffer hoseCodeBuffer = new StringBuffer();
					  StringTokenizer st = new StringTokenizer(hosecode, "()/");
					  for (int l = 0; l < k; l++) {
						  if (st.hasMoreTokens()) {
							  String partcode = st.nextToken();
							  hoseCodeBuffer.append(partcode);
						  }
						  if (l == 0) {
							  hoseCodeBuffer.append("(");
						  } else if (l == 3) {
							  hoseCodeBuffer.append(")");
						  } else {
							  hoseCodeBuffer.append("/");
						  }
					  }
					  hosecode = hoseCodeBuffer.toString();
					  String sql="select * from ATOM where HOSE_CODE like \""+hosecode+"%\";";
					  ResultSet rs=stmt.executeQuery(sql);
					  if(!rs.next()){
						  notfound[k]++;
						  if(k==1 && wishedstructures<10){
							  wishit=true;
						  }
					  }else{
						  found[k]++;
					  }
				  }
			  }
		  }
		  System.err.println(ncicount);
		  for(int i=1;i<7;i++)
			  System.err.println("Spheres; "+i+", found:"+found[i]+", not found:"+notfound[i]);
